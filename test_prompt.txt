#### TESTING 01 #### C++ Bellman-Ford Implementation (Negative Weights)

void bellmanFord(int V, int E, int src, vector<Edge>& edges) {
    vector<int> dist(V, 1e9);
    dist[src] = 0;

    for (int i = 1; i <= V - 1; i++) {
        for (int j = 0; j < E; j++) {
            int u = edges[j].u;
            int v = edges[j].v;
            int w = edges[j].weight;
            if (dist[u] != 1e9 && dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
            }
        }
    }

    for (int i = 0; i < E; i++) {
        int u = edges[i].u;
        int v = edges[i].v;
        int w = edges[i].weight;
        if (dist[u] != 1e9 && dist[u] + w < dist[v]) {
            cout << "Graph contains negative weight cycle" << endl;
            return;
        }
    }
}

#### TESTING 02 #### Rust Kruskal Implementation (Disjoint Set Union)

struct UnionFind {
    parent: Vec<usize>,
}

impl UnionFind {
    fn new(n: usize) -> Self {
        UnionFind { parent: (0..n).collect() }
    }
    fn find(&mut self, i: usize) -> usize {
        if self.parent[i] == i { i }
        else {
            self.parent[i] = self.find(self.parent[i]);
            self.parent[i]
        }
    }
    fn union(&mut self, i: usize, j: usize) -> bool {
        let root_i = self.find(i);
        let root_j = self.find(j);
        if root_i != root_j {
            self.parent[root_i] = root_j;
            true
        } else {
            false
        }
    }
}

pub fn kruskal(v: usize, mut edges: Vec<(usize, usize, i32)>) -> i32 {
    edges.sort_by_key(|k| k.2);
    let mut uf = UnionFind::new(v);
    let mut mst_weight = 0;
    for (u, v, w) in edges {
        if uf.union(u, v) {
            mst_weight += w;
        }
    }
    mst_weight
}

#### TESTING 03 #### Go PageRank Implementation (Iterative Score Vector)

package main

func calculatePageRank(nodes []string, adjMap map[string][]string, damping float64, iterations int) map[string]float64 {
	N := float64(len(nodes))
	ranks := make(map[string]float64)
	for _, node := range nodes {
		ranks[node] = 1.0 / N
	}

	for it := 0; it < iterations; it++ {
		newRanks := make(map[string]float64)
		for _, node := range nodes {
			newRanks[node] = (1.0 - damping) / N
		}
		for u, neighbors := range adjMap {
			if len(neighbors) > 0 {
				share := ranks[u] / float64(len(neighbors))
				for _, v := range neighbors {
					newRanks[v] += damping * share
				}
			}
		}
		ranks = newRanks
	}
	return ranks
}

function dijkstra(graph, start) {
    const distances = {};
    const visited = new Set();
    const pq = new MinPriorityQueue({ priority: x => x.dist });

    for (const node in graph) distances[node] = Infinity;
    distances[start] = 0;
    pq.enqueue({ node: start, dist: 0 });

    while (!pq.isEmpty()) {
        const { node, dist } = pq.dequeue();
        if (visited.has(node)) continue;
        visited.add(node);

        for (const [neighbor, weight] of graph[node]) {
            const newDist = dist + weight;
            if (newDist < distances[neighbor]) {
                distances[neighbor] = newDist;
                pq.enqueue({ node: neighbor, dist: newDist });
            }
        }
    }
    return distances;
}
